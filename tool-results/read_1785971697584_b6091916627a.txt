   108→async function api(path: string, opts: { method?: string; body?: any } = {}) {
   109→  const res = await fetch(path, {
   110→    method: opts.method || 'POST',
   111→    headers: { 'Content-Type': 'application/json' },
   112→    body: opts.body ? JSON.stringify(opts.body) : undefined,
   113→  })
   114→  if (!res.ok) throw new Error((await res.json().catch(() => ({}))).error || 'Request failed')
   115→  return res.json()
   116→}

   234→export function SmartMap() {
   235→  const city = useAppStore((s) => s.city)
   236→  const [selectedCats, setSelectedCats] = React.useState<string[]>(['restaurant', 'hospital', 'metro'])
   237→  const [budgetFilter, setBudgetFilter] = React.useState<BudgetFilter>('all')
   238→  const [ratingFilter, setRatingFilter] = React.useState<RatingFilter>('all')
   239→  const [openOnly, setOpenOnly] = React.useState(false)
   240→  const [favoritesOnly, setFavoritesOnly] = React.useState(false)
   241→  const [activeId, setActiveId] = React.useState<string | null>(null)
   242→  const [savedIds, setSavedIds] = React.useState<Set<string>>(new Set())
   243→  const [hoveredId, setHoveredId] = React.useState<string | null>(null)
   244→  const [savingId, setSavingId] = React.useState<string | null>(null)
   245→
   246→  const toggleCat = (id: string) => {
   247→    setSelectedCats((prev) =>
   248→      prev.includes(id) ? prev.filter((c) => c !== id) : [...prev, id]
   249→    )
   250→  }
   251→
   252→  const filtered = React.useMemo(() => {
   253→    return PLACES.filter((p) => {
   254→      if (!selectedCats.includes(p.category)) return false
   255→      if (budgetFilter === 'budget' && p.premium) return false
   256→      if (budgetFilter === 'premium' && !p.premium) return false
   257→      if (ratingFilter === '4+' && p.rating < 4) return false
   258→      if (ratingFilter === '4.5+' && p.rating < 4.5) return false
   259→      if (openOnly && !p.open) return false
   260→      if (favoritesOnly && !savedIds.has(p.id)) return false
   261→      return true
   262→    })
   263→  }, [selectedCats, budgetFilter, ratingFilter, openOnly, favoritesOnly, savedIds])
   264→
   265→  const handleSave = async (place: Place) => {
   266→    if (savedIds.has(place.id)) return
   267→    setSavingId(place.id)
   268→    try {
   269→      await api('/api/places', {
   270→        body: {
   271→          name: place.name,
   272→          category: place.category,
   273→          address: place.address,
   274→          rating: place.rating,
   275→          price: place.price,
   276→          distance: place.distance,
   277→          notes: `Saved from Smart Map · ${city}`,
   278→        },
   279→      })
   280→      setSavedIds((prev) => new Set(prev).add(place.id))
   281→      toast.success(`Saved "${place.name}" to your places`)
   282→    } catch (err) {
   283→      toast.error(err instanceof Error ? err.message : 'Failed to save place')
   284→    } finally {
   285→      setSavingId(null)
   286→    }
   287→  }
   288→
   289→  const handleSaveAll = async () => {
   290→    const toSave = filtered.filter((p) => !savedIds.has(p.id))
   291→    if (toSave.length === 0) {
   292→      toast.info('All visible places are already saved')
   293→      return
   294→    }
   295→    toast.success(`Saving ${toSave.length} places...`)
   296→    let ok = 0
   297→    for (const p of toSave) {
   298→      try {
   299→        await handleSave(p)
   300→        ok++
   301→      } catch {
   302→        // handled in handleSave
   303→      }
   304→    }
   305→    if (ok > 0) toast.success(`Saved ${ok} places`)
   306→  }
   307→
   308→  return (
   309→    <div className="p-4 lg:p-6 max-w-7xl mx-auto">
   310→      <div className="flex items-start justify-between flex-wrap gap-3 mb-4">
   311→        <div>
   312→          <h1 className="text-xl sm:text-2xl font-bold tracking-tight flex items-center gap-2">
   313→            <MapPin className="size-5 text-emerald-600" />
   314→            Smart Map
   315→          </h1>
   316→          <p className="text-sm text-muted-foreground mt-0.5">
   317→            Explore essentials around <span className="font-medium text-foreground">{city}</span>
   318→          </p>
   319→        </div>
   320→        <Button
   321→          variant="outline"
   322→          size="sm"
   323→          onClick={handleSaveAll}
   324→          className="border-emerald-500/40 text-emerald-700 dark:text-emerald-400 hover:bg-emerald-500/10"
   325→        >
   326→          <BookmarkPlus className="size-4" />
   327→          Save all visible ({filtered.filter((p) => !savedIds.has(p.id)).length})
   328→        </Button>
   329→      </div>
   330→
   331→      {/* Category chips */}
   332→      <Card className="p-3 mb-4 gap-0">
   333→        <div className="flex items-center gap-1.5 mb-2">
   334→          <Filter className="size-3.5 text-muted-foreground" />
   335→          <span className="text-xs font-semibold text-muted-foreground">Categories</span>
   336→        </div>
   337→        <div className="flex flex-wrap gap-2">
   338→          {PLACE_CATEGORIES.map((c) => {
   339→            const Icon = ICONS[c.icon] || MapPin
   340→            const active = selectedCats.includes(c.id)
   341→            const col = colorFor(c.id)
   342→            return (
   343→              <button
   344→                key={c.id}
   345→                onClick={() => toggleCat(c.id)}
   346→                className={cn(
   347→                  'inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-medium border transition-all',
   348→                  active
   349→                    ? cn('border-transparent text-white shadow-sm', col.pin)
   350→                    : 'bg-background hover:bg-accent text-muted-foreground hover:text-foreground'
   351→                )}
   352→              >
   353→                <Icon className="size-3.5" />
   354→                {c.label}
   355→              </button>
   356→            )
   357→          })}
   358→        </div>
   359→      </Card>
   360→
   361→      <div className="grid lg:grid-cols-[1fr_340px] gap-4">
   362→        {/* Map area */}
   363→        <div className="flex flex-col gap-3 order-1">
   364→          {/* Filters bar */}
   365→          <Card className="p-3 gap-0">
   366→            <div className="flex items-center gap-3 flex-wrap">
   367→              <div className="flex items-center gap-1.5">
   368→                <span className="text-[11px] text-muted-foreground mr-1">Budget</span>
   369→                {(['all', 'budget', 'premium'] as BudgetFilter[]).map((b) => (
   370→                  <button
   371→                    key={b}
   372→                    onClick={() => setBudgetFilter(b)}
   373→                    className={cn(
   374→                      'text-xs px-2.5 py-1 rounded-md font-medium capitalize transition-colors',
   375→                      budgetFilter === b
   376→                        ? 'bg-emerald-600 text-white'
   377→                        : 'bg-muted/50 text-muted-foreground hover:bg-muted'
   378→                    )}
   379→                  >
   380→                    {b === 'all' ? 'All' : b}
   381→                  </button>
   382→                ))}
   383→              </div>
   384→
   385→              <div className="h-5 w-px bg-border" />
   386→
   387→              <div className="flex items-center gap-1.5">
   388→                <span className="text-[11px] text-muted-foreground mr-1">Rating</span>
   389→                {(['all', '4+', '4.5+'] as RatingFilter[]).map((r) => (
   390→                  <button
   391→                    key={r}
   392→                    onClick={() => setRatingFilter(r)}
   393→                    className={cn(
   394→                      'text-xs px-2.5 py-1 rounded-md font-medium transition-colors',
   395→                      ratingFilter === r
   396→                        ? 'bg-amber-500 text-white'
   397→                        : 'bg-muted/50 text-muted-foreground hover:bg-muted'
   398→                    )}
   399→                  >
   400→                    {r === 'all' ? 'All' : `${r}★`}
